2. Now make the 🐍 Snake actually work

The README above expects this file:

.github/workflows/snake.yml

Create that file in the same Sanjana-kumaresan repository and paste:

name: 🐍 Generate Contribution Snake

on:
  schedule:
    - cron: "0 0 * * *"
  workflow_dispatch:

jobs:
  generate:
    runs-on: ubuntu-latest

    permissions:
      contents: write

    steps:
      - name: 🐍 Generate Snake
        uses: Platane/snk@v3
        with:
          github_user_name: Sanjana-kumaresan
          outputs: |
            dist/github-contribution-grid-snake.svg
            dist/github-contribution-grid-snake-dark.svg?palette=github-dark

      - name: 🚀 Publish Snake
        uses: crazy-max/ghaction-github-pages@v4
        with:
          build_dir: dist
        env:
          GITHUB_TOKEN: ${{ secrets.GITHUB_TOKEN }}
          BUILD_DIR: dist
